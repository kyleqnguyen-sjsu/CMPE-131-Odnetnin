
using UnityEngine;
using UnityEngine.InputSystem;

public class KartController : MonoBehaviour
{
    // =====================================
    // WHEEL COLLIDERS
    // =====================================

    public WheelCollider frontLeftWheel;
    public WheelCollider frontRightWheel;
    public WheelCollider rearLeftWheel;
    public WheelCollider rearRightWheel;

    // =====================================
    // WHEEL MESHES
    // =====================================

    public Transform frontLeftMesh;
    public Transform frontRightMesh;
    public Transform rearLeftMesh;
    public Transform rearRightMesh;

    // =====================================
    // ENGINE
    // =====================================

    public float motorTorque = 4000f;
    public float maxSpeed = 45f;
    public float reverseSpeed = 25f;

    public float arcadeAcceleration = 65f;
    public float reverseAcceleration = 18f;
    public float speedTaperStart = 0.9f;

    // =====================================
    // PROGRESSIVE GAS
    // =====================================

    public float throttleRampTime = 1.5f;
    public float throttleReleaseTime = 0.25f;

    // =====================================
    // BRAKING
    // =====================================

    public float brakeForce = 8000f;
    public float arcadeBrakeDeceleration = 100f;
    public float brakeRampTime = 0.3f;
    public float brakeReleaseTime = 0.15f;

    public float idleBrakeForce = 600f;
    public float bothKeysBrakeForce = 8000f;
    public float bothKeysArcadeDeceleration = 50f;

    // =====================================
    // DIRECTION
    // =====================================

    public float stopThreshold = 0.5f;
    public float directionChangeDelay = 0.1f;

    // =====================================
    // NORMAL STEERING
    // =====================================

    public float maxSteerAngle = 26f;
    public float steeringResponse = 240f;

    // =====================================
    // DRIFT STEERING
    // =====================================

    public float minimumDriftSpeed = 15f;
    public float driftSteerAngle = 36f;

    public float driftBuildSpeed = 0.7f;
    public float counterSteerRecovery = 2.2f;

    // =====================================
    // DRIFT GRIP
    // =====================================

    public float driftGrip = 0.90f;
    public float minimumDriftGrip = 0.78f;
    public float driftRecoverySpeed = 2.5f;

    // =====================================
    // DRIFT POWER
    // =====================================

    public float driftPowerMultiplier = 0.20f;

    // =====================================
    // DRIFT SPEED LOSS
    // =====================================

    public float driftBaseDrag = 5f;
    public float driftSlipDrag = 8f;
    public float driftMinimumSpeed = 8f;

    // =====================================
    // DRIFT STABILITY
    // =====================================

    public float lateralGripAssist = 5f;
    public float maxLateralAssist = 22f;

    public float maxDriftSlipAngle = 25f;
    public float yawStability = 3f;

    // =====================================
    // DRIFT EXIT
    // =====================================

    public float driftExitDuration = 1.2f;
    public float driftExitTorqueMultiplier = 1.1f;

    // =====================================
    // DEBUG
    // =====================================

    public bool showDriftDebug = false;

    // =====================================
    // PRIVATE VARIABLES
    // =====================================

    private Rigidbody rb;

    private bool gas;
    private bool brake;
    private bool drift;
    private float turn;

    private float throttleInput = 0f;
    private float brakeInput = 0f;
    private float reverseThrottle = 0f;

    private bool isDrifting = false;

    private float driftDirection = 0f;
    private float driftTightness = 0f;
    private float driftExitTimer = 0f;

    private float changeTimer = 0f;
    private float currentSteerAngle = 0f;
    private float currentGrip = 1f;

    private float rearLeftNormalGrip;
    private float rearRightNormalGrip;

    private float currentSlipAngle = 0f;
    private float rearWheelSlip = 0f;

    private float debugTimer = 0f;

    // =====================================
    // START
    // =====================================

    void Start()
    {
        rb = GetComponent<Rigidbody>();

        if (rb == null ||
            frontLeftWheel == null ||
            frontRightWheel == null ||
            rearLeftWheel == null ||
            rearRightWheel == null)
        {
            Debug.LogError(
                "Missing Rigidbody or WheelCollider!"
            );

            enabled = false;
            return;
        }

        rearLeftNormalGrip =
            rearLeftWheel.sidewaysFriction.stiffness;

        rearRightNormalGrip =
            rearRightWheel.sidewaysFriction.stiffness;
    }

    // =====================================
    // INPUT
    // =====================================

    void Update()
    {
        if (Keyboard.current == null)
        {
            gas = false;
            brake = false;
            drift = false;
            turn = 0f;
            return;
        }

        gas = Keyboard.current.wKey.isPressed;
        brake = Keyboard.current.sKey.isPressed;
        drift = Keyboard.current.spaceKey.isPressed;

        bool left = Keyboard.current.aKey.isPressed;
        bool right = Keyboard.current.dKey.isPressed;

        turn = 0f;

        if (left && !right)
            turn = -1f;

        else if (right && !left)
            turn = 1f;
    }

    // =====================================
    // PHYSICS
    // =====================================

    void FixedUpdate()
    {
        if (rb == null)
            return;

        float dt = Time.fixedDeltaTime;

        Vector3 localVelocity =
            transform.InverseTransformDirection(
                rb.linearVelocity
            );

        float forwardSpeed = localVelocity.z;

        Vector3 groundVelocity =
            Vector3.ProjectOnPlane(
                rb.linearVelocity,
                transform.up
            );

        float actualSpeed = groundVelocity.magnitude;

        // =====================================
        // CALCULATE BODY SLIP ANGLE
        // =====================================

        if (actualSpeed > 1f)
        {
            currentSlipAngle =
                Mathf.Atan2(
                    localVelocity.x,
                    Mathf.Abs(localVelocity.z)
                ) * Mathf.Rad2Deg;
        }
        else
        {
            currentSlipAngle = 0f;
        }

        // =====================================
        // MEASURE REAR WHEEL SLIP
        // =====================================

        rearWheelSlip = GetRearWheelSlip();

        // =====================================
        // DRIFT STATE
        // =====================================

        bool wasDrifting = isDrifting;

        if (!drift || actualSpeed < 5f)
        {
            isDrifting = false;
        }
        else if (!isDrifting &&
                 forwardSpeed >= minimumDriftSpeed &&
                 Mathf.Abs(turn) > 0.1f)
        {
            isDrifting = true;

            driftDirection = Mathf.Sign(turn);
            driftTightness = 0.2f;
        }

        if (wasDrifting && !isDrifting)
        {
            driftExitTimer = driftExitDuration;

            driftDirection = 0f;
            driftTightness = 0f;
        }

        if (isDrifting)
            driftExitTimer = 0f;

        // =====================================
        // PROGRESSIVE DRIFT
        // =====================================

        if (isDrifting)
        {
            if (Mathf.Abs(turn) > 0.1f)
            {
                if (turn == driftDirection)
                {
                    // Tighten drift
                    driftTightness +=
                        driftBuildSpeed * dt;
                }
                else
                {
                    // Countersteer
                    driftTightness -=
                        counterSteerRecovery * dt;

                    if (driftTightness <= 0f)
                    {
                        driftDirection =
                            Mathf.Sign(turn);

                        driftTightness = 0.1f;
                    }
                }
            }
            else
            {
                driftTightness -= 0.15f * dt;
            }

            driftTightness =
                Mathf.Clamp01(driftTightness);
        }

        // =====================================
        // FRONT WHEEL STEERING
        // =====================================

        float steerLimit =
            isDrifting
            ? Mathf.Lerp(
                maxSteerAngle,
                driftSteerAngle,
                driftTightness
              )
            : maxSteerAngle;

        currentSteerAngle = Mathf.MoveTowards(
            currentSteerAngle,
            turn * steerLimit,
            steeringResponse * dt
        );

        frontLeftWheel.steerAngle =
            currentSteerAngle;

        frontRightWheel.steerAngle =
            currentSteerAngle;

        // =====================================
        // REAR WHEEL GRIP
        // =====================================

        float targetGrip = 1f;

        if (isDrifting)
        {
            targetGrip = Mathf.Lerp(
                driftGrip,
                minimumDriftGrip,
                driftTightness
            );
        }

        currentGrip = Mathf.MoveTowards(
            currentGrip,
            targetGrip,
            driftRecoverySpeed * dt
        );

        SetRearGrip(currentGrip);

        // =====================================
        // INPUT STATES
        // =====================================

        bool bothKeys = gas && brake;
        bool forwardRequest = gas && !brake;
        bool reverseRequest = brake && !gas;

        bool recovering =
            isDrifting ||
            driftExitTimer > 0f ||
            currentGrip < 0.98f;

        bool sidewaysSliding =
            Mathf.Abs(localVelocity.x) >
            Mathf.Abs(forwardSpeed) * 0.5f;

        bool movingBackward =
            forwardSpeed < -stopThreshold &&
            !sidewaysSliding &&
            !recovering;

        bool needsForwardBraking =
            forwardRequest && movingBackward;

        bool movingForwardOrSliding =
            actualSpeed > stopThreshold &&
            forwardSpeed >= -stopThreshold;

        bool needsReverseBraking =
            reverseRequest && movingForwardOrSliding;

        bool activeBraking =
            needsForwardBraking ||
            needsReverseBraking;

        // =====================================
        // PROGRESSIVE GAS AND BRAKE
        // =====================================

        float targetThrottle =
            forwardRequest && !needsForwardBraking
            ? 1f : 0f;

        float targetBrake =
            activeBraking ? 1f : 0f;

        float throttleTime =
            targetThrottle > throttleInput
            ? throttleRampTime
            : throttleReleaseTime;

        float brakeTime =
            targetBrake > brakeInput
            ? brakeRampTime
            : brakeReleaseTime;

        throttleInput = Mathf.MoveTowards(
            throttleInput,
            targetThrottle,
            dt / Mathf.Max(0.01f, throttleTime)
        );

        brakeInput = Mathf.MoveTowards(
            brakeInput,
            targetBrake,
            dt / Mathf.Max(0.01f, brakeTime)
        );

        // =====================================
        // ENGINE AND BRAKES
        // =====================================

        float torque = 0f;
        float braking = 0f;
        float acceleration = 0f;

        // W + S
        if (bothKeys)
        {
            throttleInput = 0f;
            brakeInput = 0f;
            reverseThrottle = 0f;

            braking = bothKeysBrakeForce;

            ApplyArcadeBraking(
                bothKeysArcadeDeceleration,
                dt
            );

            changeTimer = 0f;
        }

        // Brake before switching direction
        else if (activeBraking)
        {
            throttleInput = 0f;
            reverseThrottle = 0f;

            braking = brakeForce * brakeInput;

            ApplyArcadeBraking(
                arcadeBrakeDeceleration * brakeInput,
                dt
            );

            changeTimer = 0f;
        }

        // Forward driving
        else if (forwardRequest)
        {
            reverseThrottle = 0f;
            changeTimer += dt;

            if (changeTimer >= directionChangeDelay ||
                recovering)
            {
                float powerFactor =
                    GetPowerFactor(
                        actualSpeed,
                        maxSpeed
                    );

                torque =
                    motorTorque *
                    throttleInput *
                    powerFactor;

                acceleration =
                    arcadeAcceleration *
                    throttleInput *
                    powerFactor;

                if (isDrifting)
                {
                    // Wheels still receive some power
                    torque *= driftPowerMultiplier;

                    // No extra Rigidbody acceleration
                    // while drifting
                    acceleration = 0f;
                }

                if (driftExitTimer > 0f)
                {
                    torque *=
                        driftExitTorqueMultiplier;
                }
            }
        }

        // Reverse driving
        else if (reverseRequest)
        {
            throttleInput = 0f;
            changeTimer += dt;

            if (actualSpeed <= stopThreshold ||
                forwardSpeed < -stopThreshold)
            {
                reverseThrottle = Mathf.MoveTowards(
                    reverseThrottle,
                    1f,
                    dt / Mathf.Max(
                        0.01f,
                        throttleRampTime
                    )
                );

                if (changeTimer >= directionChangeDelay)
                {
                    float reverseFactor =
                        GetPowerFactor(
                            Mathf.Max(0f, -forwardSpeed),
                            reverseSpeed
                        );

                    torque =
                        -motorTorque *
                        0.6f *
                        reverseThrottle *
                        reverseFactor;

                    acceleration =
                        -reverseAcceleration *
                        reverseThrottle *
                        reverseFactor;
                }
            }
        }

        // No W or S
        else
        {
            reverseThrottle = 0f;
            braking = idleBrakeForce;
            changeTimer = 0f;
        }

        // =====================================
        // ARCADE ACCELERATION
        // =====================================

        if (acceleration > 0f)
        {
            Vector3 driveDirection =
                transform.forward;

            if (recovering && actualSpeed > 1f)
            {
                driveDirection = Vector3.Slerp(
                    groundVelocity.normalized,
                    transform.forward,
                    0.2f
                ).normalized;
            }

            rb.AddForce(
                driveDirection * acceleration,
                ForceMode.Acceleration
            );
        }
        else if (acceleration < 0f)
        {
            rb.AddForce(
                transform.forward * acceleration,
                ForceMode.Acceleration
            );
        }

        // =====================================
        // NEW: SLIP-BASED DRIFT PHYSICS
        // =====================================

        if (isDrifting &&
            !bothKeys &&
            !activeBraking)
        {
            ApplyDriftPhysics(
                actualSpeed,
                localVelocity,
                groundVelocity,
                dt
            );
        }

        // =====================================
        // APPLY WHEEL TORQUE AND BRAKES
        // =====================================

        rearLeftWheel.motorTorque = torque;
        rearRightWheel.motorTorque = torque;

        frontLeftWheel.brakeTorque = braking;
        frontRightWheel.brakeTorque = braking;
        rearLeftWheel.brakeTorque = braking;
        rearRightWheel.brakeTorque = braking;

        if (driftExitTimer > 0f)
        {
            driftExitTimer = Mathf.Max(
                0f,
                driftExitTimer - dt
            );
        }

        // =====================================
        // OPTIONAL DEBUG
        // =====================================

        if (showDriftDebug)
        {
            debugTimer += dt;

            if (debugTimer >= 0.5f)
            {
                debugTimer = 0f;

                Debug.Log(
                    "Speed: " +
                    actualSpeed.ToString("F1") +
                    " | Drifting: " + isDrifting +
                    " | Slip Angle: " +
                    currentSlipAngle.ToString("F1") +
                    " | Rear Slip: " +
                    rearWheelSlip.ToString("F2") +
                    " | Grip: " +
                    currentGrip.ToString("F2")
                );
            }
        }
    }

    // =====================================
    // READ REAR WHEEL SLIP
    // =====================================

    float GetRearWheelSlip()
    {
        float totalSlip = 0f;
        int groundedWheels = 0;

        WheelHit hit;

        if (rearLeftWheel.GetGroundHit(out hit))
        {
            totalSlip +=
                Mathf.Abs(hit.sidewaysSlip);

            groundedWheels++;
        }

        if (rearRightWheel.GetGroundHit(out hit))
        {
            totalSlip +=
                Mathf.Abs(hit.sidewaysSlip);

            groundedWheels++;
        }

        if (groundedWheels == 0)
            return 0f;

        return totalSlip / groundedWheels;
    }

    // =====================================
    // SLIP-BASED DRIFT PHYSICS
    // =====================================

    void ApplyDriftPhysics(
        float speed,
        Vector3 localVelocity,
        Vector3 groundVelocity,
        float dt
    )
    {
        if (speed < 1f)
            return;

        // =================================
        // 1. MEASURE DRIFT INTENSITY
        // =================================

        float bodySlipFactor =
            Mathf.Clamp01(
                Mathf.Abs(currentSlipAngle) /
                Mathf.Max(1f, maxDriftSlipAngle)
            );

        float wheelSlipFactor =
            Mathf.Clamp01(
                rearWheelSlip / 0.8f
            );

        float driftIntensity =
            Mathf.Max(
                bodySlipFactor,
                wheelSlipFactor
            );

        // =================================
        // 2. DRIFT SPEED LOSS
        // =================================

        if (speed > driftMinimumSpeed)
        {
            float drag =
                driftBaseDrag +
                driftSlipDrag * driftIntensity;

            // Do not create excessive braking
            // close to the minimum speed.
            float speedFactor =
                Mathf.Clamp01(
                    (speed - driftMinimumSpeed) /
                    5f
                );

            rb.AddForce(
                -groundVelocity.normalized *
                drag * speedFactor,
                ForceMode.Acceleration
            );
        }

        // =================================
        // 3. LATERAL STABILITY ASSIST
        // =================================

        // Sideways velocity relative to car
        float sidewaysSpeed = localVelocity.x;

        // Counter excessive sideways motion
        float lateralCorrection =
            -sidewaysSpeed * lateralGripAssist;

        lateralCorrection = Mathf.Clamp(
            lateralCorrection,
            -maxLateralAssist,
            maxLateralAssist
        );

        rb.AddForce(
            transform.right * lateralCorrection,
            ForceMode.Acceleration
        );

        // =================================
        // 4. ANTI-SPIN ASSIST
        // =================================

        float excessSlip =
            Mathf.Clamp01(
                (Mathf.Abs(currentSlipAngle) -
                 maxDriftSlipAngle) / 15f
            );

        if (excessSlip > 0f)
        {
            Vector3 angularVelocity =
                rb.angularVelocity;

            Vector3 yawAxis = transform.up;

            float yawSpeed =
                Vector3.Dot(
                    angularVelocity,
                    yawAxis
                );

            rb.AddTorque(
                -yawAxis *
                yawSpeed *
                yawStability *
                excessSlip,
                ForceMode.Acceleration
            );
        }
    }

    // =====================================
    // ARCADE BRAKING
    // =====================================

    void ApplyArcadeBraking(
        float deceleration,
        float dt
    )
    {
        Vector3 groundVelocity =
            Vector3.ProjectOnPlane(
                rb.linearVelocity,
                transform.up
            );

        Vector3 newVelocity =
            Vector3.MoveTowards(
                groundVelocity,
                Vector3.zero,
                Mathf.Max(0f, deceleration) * dt
            );

        rb.linearVelocity +=
            newVelocity - groundVelocity;
    }

    // =====================================
    // ENGINE SPEED TAPER
    // =====================================

    float GetPowerFactor(float speed, float limit)
    {
        if (limit <= 0f)
            return 0f;

        float start =
            limit * Mathf.Clamp01(speedTaperStart);

        if (speed <= start)
            return 1f;

        if (speed >= limit)
            return 0f;

        float t =
            Mathf.InverseLerp(start, limit, speed);

        return 1f -
            Mathf.SmoothStep(0f, 1f, t);
    }

    // =====================================
    // WHEEL MESH ANIMATION
    // =====================================

    void LateUpdate()
    {
        UpdateWheel(frontLeftWheel, frontLeftMesh);
        UpdateWheel(frontRightWheel, frontRightMesh);
        UpdateWheel(rearLeftWheel, rearLeftMesh);
        UpdateWheel(rearRightWheel, rearRightMesh);
    }

    void UpdateWheel(
        WheelCollider wheel,
        Transform wheelMesh
    )
    {
        if (wheel == null || wheelMesh == null)
            return;

        wheel.GetWorldPose(
            out Vector3 position,
            out Quaternion rotation
        );

        wheelMesh.position = position;
        wheelMesh.rotation = rotation;
    }

    // =====================================
    // REAR TIRE GRIP
    // =====================================

    void SetRearGrip(float multiplier)
    {
        WheelFrictionCurve left =
            rearLeftWheel.sidewaysFriction;

        WheelFrictionCurve right =
            rearRightWheel.sidewaysFriction;

        left.stiffness =
            rearLeftNormalGrip * multiplier;

        right.stiffness =
            rearRightNormalGrip * multiplier;

        rearLeftWheel.sidewaysFriction = left;
        rearRightWheel.sidewaysFriction = right;
    }

    // =====================================
    // RESET GRIP
    // =====================================

    void OnDisable()
    {
        if (rearLeftWheel != null &&
            rearLeftNormalGrip > 0f)
        {
            WheelFrictionCurve left =
                rearLeftWheel.sidewaysFriction;

            left.stiffness = rearLeftNormalGrip;
            rearLeftWheel.sidewaysFriction = left;
        }

        if (rearRightWheel != null &&
            rearRightNormalGrip > 0f)
        {
            WheelFrictionCurve right =
                rearRightWheel.sidewaysFriction;

            right.stiffness = rearRightNormalGrip;
            rearRightWheel.sidewaysFriction = right;
        }
    }
}
